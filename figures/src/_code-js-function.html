// ฟังก์ชันและ callback ใน progress.js — แก้สถานะผ่านฟังก์ชันแล้วบันทึกอัตโนมัติ
function update(fn) {
  var state = load();
  fn(state);        // เรียก callback ที่ผู้ใช้ส่งเข้ามาแก้ไขสถานะ
  save(state);
  return state;
}

// การใช้งาน: ส่งฟังก์ชันเข้าไปเป็นพารามิเตอร์
